% ---------------------------------------------------------------- boxes
% keyidea: the one thing to remember.   pitfall: the usual way to get it wrong.
% Both are unbreakable, so keep them to a few lines.
\newsavebox{\calbox}
\newenvironment{calframed}[2]{%
  \def\calfill{#1}\par\medskip\noindent
  \begin{lrbox}{\calbox}\begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
  \setlength{\parskip}{0.4em}\textbf{#2}\ }{%
  \end{minipage}\end{lrbox}\fcolorbox{calrule}{\calfill}{\usebox{\calbox}}\par\medskip}
\newenvironment{keyidea}{\begin{calframed}{calidea}{Key idea.}}{\end{calframed}}
\newenvironment{pitfall}{\begin{calframed}{calwarn}{Pitfall.}}{\end{calframed}}
